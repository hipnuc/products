cmake_minimum_required(VERSION 3.16)
project(hipnuc_stm32_receive_tests C)
enable_testing()
add_executable(test_serial_dma test_serial_dma.c)
add_executable(test_serial_irq test_serial_irq.c)
add_executable(test_can_queue test_can_queue.c)
add_executable(test_serial_application test_application.c)
add_executable(test_can_application test_application.c)
target_compile_definitions(test_can_application PRIVATE TEST_CAN_APPLICATION)
foreach(target test_serial_dma test_serial_irq test_can_queue test_serial_application test_can_application)
  target_compile_features(${target} PRIVATE c_std_99)
  target_include_directories(${target} PRIVATE stubs ../../c/hipnuc)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
