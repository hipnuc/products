cmake_minimum_required(VERSION 3.16)
project(hipnuc_examples C)
if(NOT TARGET hipnuc_serial)
    add_subdirectory(../serial serial)
endif()
add_executable(read_c read.c)
target_link_libraries(read_c PRIVATE hipnuc_serial)
set_target_properties(read_c PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)

option(HIPNUC_BUILD_CPP_EXAMPLE "Build the C++ example when a compiler is available" ON)
if(HIPNUC_BUILD_CPP_EXAMPLE)
    include(CheckLanguage)
    check_language(CXX)
    if(CMAKE_CXX_COMPILER)
        enable_language(CXX)
        add_executable(read_cpp read.cpp)
        target_link_libraries(read_cpp PRIVATE hipnuc_serial)
        set_target_properties(read_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
    else()
        message(STATUS "C++ compiler unavailable; building C examples only")
    endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(read_can read_can.c)
    target_link_libraries(read_can PRIVATE hipnuc_j1939)
    set_target_properties(read_can PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
endif()
